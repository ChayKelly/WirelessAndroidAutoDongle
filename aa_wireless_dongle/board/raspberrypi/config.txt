kernel=zImage

disable_splash=1
boot_delay=0
disable_poe_fan=1

# To use an external initramfs file
#initramfs rootfs.cpio.gz

# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1
# Add the following line if the board need reduce power drain and increase performance disabling hdmi port
hdmi_blanking=1

# How much memory in MB to assign to the GPU on Pi models having
# 256, 512 or 1024 MB total memory
gpu_mem_256=100
gpu_mem_512=100
gpu_mem_1024=100

# Enable autoprobing of Bluetooth driver without need of hciattach/btattach
dtparam=krnbt=on

# Enable dwc2 required for gadget mode
dtoverlay=dwc2,dr_mode=peripheral

# Hold the ARM clock at its rated speed instead of letting it idle at 600MHz.
# Drive 10 (20 Sep 2026) read freq=600M on all 4,815 heartbeat samples, including
# sessions carrying 600 kB/s through dwc2 and the wifi, so the governor never
# raised it under load. The proxy, the USB interrupt path and hostapd all share
# that clock, and every cycle it loses is jitter on the audio. Costs nothing on a
# board that peaks at 53C (no throttling or under-voltage recorded on any drive).
# arm_freq_min rather than force_turbo: it leaves the firmware's thermal limits
# in place and does not set the warranty bit.
arm_freq_min=1000
